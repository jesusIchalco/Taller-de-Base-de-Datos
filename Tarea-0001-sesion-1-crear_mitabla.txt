CREATE DATABASE 
  `Jesus Chalco` CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- Seleccionar la base de datos
USE `Jesus Chalco`;

-- Crear la primera tabla para las comidas
CREATE TABLE Comidas (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  platillo VARCHAR(100) NOT NULL
);
CREATE TABLE `Lugares de viaje` (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  lugar VARCHAR(100) NOT NULL
);
